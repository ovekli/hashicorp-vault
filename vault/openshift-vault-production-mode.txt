# On openshift install vault
git clone https://github.com/hashicorp-education/learn-vault-redhat-openshift
cd learn-vault-redhat-openshift
#oc create namespace vault
helm repo add hashicorp https://helm.releases.hashicorp.com
helm repo update
helm install vault hashicorp/vault \
    --create-namespace \
    --namespace vault \
    --set "global.openshift=true" \
    --set "server.dev.enabled=true" \
    --set "server.image.repository=docker.io/hashicorp/vault" \
    --set "injector.image.repository=docker.io/hashicorp/vault-k8s"

# Switch from den (in-memory-storage, dev-mode to file-storage, production-mode)
oc create -f - << EOF
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: vault
spec:
  accessModes:
  - ReadWriteOnce
  resources:
    requests:
      storage: 1Gi
EOF

# Create configmap
oc create -f - << EOF
apiVersion: v1
data:
  config.hcl: "ui\t\t\t= true\ndefault_lease_ttl\t= \"168h\"\nmax_lease_ttl\t\t= \"720h\"\ndisable_mlock\t\t=
    true\n\nlistener \"tcp\" {\n  address = \"0.0.0.0:8200\"\n  tls_disable = true\n}\n\napi_addr
    = \"http://127.0.0.1:8200\"\n\nstorage \"file\" {\n  path = \"/var/lib/vault/data\"\n}\n"
kind: ConfigMap
metadata:
  name: vault
EOF

# Mount volumes to statefulsets
oc set volumes -c vault statefulset vault --add --type pvc --name vault --claim-name vault --mount-path /var/lib/vault

# Mount configmap to containers homedir
oc set volumes -c vault statefulset vault --add --type configmap --name vault-cm --configmap-name vault --mount-path /home/vault/config

# Switch to production mode
oc patch statefulset vault --patch "$(cat <<EOF
spec:
  template:
    spec:
      containers:
        - args:
          - "/usr/local/bin/docker-entrypoint.sh vault server -config /home/vault/config/config.hcl \n"
          name: vault 
EOF
)"

# Delete existing vault-0 pod
# Check that the new pod is going up normal

# Initializing vault
oc rsh -c vault vault-0 /bin/sh -c 'vault operator init'

# Be sure to save Unseal keys and Initial root token in a safe way

# Time to unseal vault
oc rsh -c vault vault-0 /bin/sh -c 'vault operator unseal <Unseal Key 1>'
oc rsh -c vault vault-0 /bin/sh -c 'vault operator unseal <Unseal Key 2>'
oc rsh -c vault vault-0 /bin/sh -c 'vault operator unseal <Unseal Key 3>'

# Create route object in openshift
oc create route edge vault --service vault --insecure-skip-tls-verify
