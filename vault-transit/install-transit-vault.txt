# Create new openshift namespace
oc create namespace vault-transit

# Install vault-transit server with helm
helm install vault-transit hashicorp/vault \
    --set "global.openshift=true" \
    --set "server.dev.enabled=true" \
    --set "server.image.repository=docker.io/hashicorp/vault" \
    --set "injector.image.repository=docker.io/hashicorp/vault-k8s"    

# Switch from den (in-memory-storage, dev-mode to file-storage, production-mode)
oc create -f - << EOF
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: vault-transit
spec:
  accessModes:
  - ReadWriteOnce
  resources:
    requests:
      storage: 1Gi
EOF

# Create configmap
oc create -f - << EOF
apiVersion: v1
data:
  config.hcl: "ui\t\t\t= true\ndefault_lease_ttl\t= \"168h\"\nmax_lease_ttl\t\t= \"720h\"\ndisable_mlock\t\t=
    true\n\nlistener \"tcp\" {\n  address = \"0.0.0.0:8200\"\n  tls_disable = true\n}\n\napi_addr
    = \"http://127.0.0.1:8200\"\n\nstorage \"file\" {\n  path = \"/var/lib/vault/data\"\n}\n"
kind: ConfigMap
metadata:
  name: vault-transit
EOF   

# Mount volumes to statefulsets
oc set volumes -c vault statefulset vault-transit --add --type pvc --name vault-transit --claim-name vault-transit --mount-path /var/lib/vault

# Mount configmap to containers homedir
oc set volumes -c vault statefulset vault-transit --add --type configmap --name vault-transit-cm --configmap-name vault-transit --mount-path /home/vault/config

# Startup pod and setup a remote shell to the container
oc rsh vault-transit-0

# In shell - initialize vault 
vault operator init

# Create an encryption key named autounseal

# Create a policy for updates
vault policy write autounseal -<<EOF
path "transit/encrypt/autounseal" {
   capabilities = [ "update" ]
}

path "transit/decrypt/autounseal" {
   capabilities = [ "update" ]
}
EOF

# Create a client token for autounseal operations
vault token create -orphan -policy="autounseal" \
   -wrap-ttl=120 -period=24h \
   -field=wrapping_token > /tmp/wrapping-token.txt
