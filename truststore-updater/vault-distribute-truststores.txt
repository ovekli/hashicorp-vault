### ON MACOS CONFIGURE OPENSHIFT VAULT SERVICE ###

# Setup envvar VAULT_ADDR
export VAULT_ADDR=https://vault-vault.apps.crc.testing

# Get minishift ingress cert with openssl s_client command and store them into a file

# Setup envvar VAULT_CACERT with path to minishift cert
export VAULT_CACERT=/Users/ovekli/ssl/minishift-ca.pem

# Create certificate
cd $HOME/work-projects/vault-store-certificates
openssl req -x509 -newkey rsa:4096 -keyout key.pem -out cert.pem -sha256 -days 365

# Create truststore
keytool -import -trustcacerts -file cert.pem -alias klingotrust -keystore klingotrust.jks

# Install vault on macos (For client use)
brew tap hashicorp/tap
brew install hashicorp/tap/vault

# Test connection 
vault status

# ALT 1 - Set envvar VAULT_TOKEN to authenticate
export VAULT_TOKEN=hvs.atkPu7DUZeroywFa1lDReVNA

# ALT 2 - Login into vault server (Add token name)
vault login
output:
Token (will be hidden):

# Create new secret - kv type - path truststores v2
vault secrets enable -path=truststores -version=2 kv

# Add truststore to secret truststores
cat klingotrust.jks | base64 |vault kv put truststores/klingotrust klingotrust=-
cat klingotrust.jks | base64 |vault kv patch truststores/klingotrust oveklitrust=-

# Try to download keystore file
vault kv get -field klingotrust truststores/klingotrust | base64 -d > vault-klingotrust.jks

# Try to list entries in trust vault-klingotrust.jks file
keytool -list -keystore vault-klingotrust.jks

### Next create pod who can login with userpass module and download the keystore file ###

# vault auth userpass create one
vault auth enable userpass

# Create user
vault write /auth/userpass/users/update-truststore-user \
    password='Get my truststores' \
    policies=truststores-policy

# List authentications
vault auth list

# Verify kubernetes auth method
vault read /auth/userpass/users/update-truststore-user

# Add read policy to secret truststores/klingotrust
vault policy write truststores-policy - <<EOF
path "truststores/+/klingotrust" {
   capabilities = ["read"]
}
EOF


### CREATE SECRET FOR VAULT USERNAME/PASSWORD ###

# create kv secret
vault secrets enable -path=secrets kv-v2

# Create kv entry username/password to vault
vault kv put secrets/truststore-updater/config username=update-truststore-user password='Get my truststores'

# Enable kubernetes auth
vault auth enable kubernetes

# Run remote shell to vault pod
oc rsh vault-0
vault write auth/kubernetes/config kubernetes_host="https://$KUBERNETES_PORT_443_TCP_ADDR:443"
vault read auth/kubernetes/config
exit

# Add read policy to secret secrets/data/truststore-updater/config
vault policy write truststore-sa - <<EOF
path "secrets/data/truststore-updater/config" {
   capabilities = ["read"]
}
EOF

# Create kubernetes auth role named truststore-sa
vault write auth/kubernetes/role/truststore-sa \
      bound_service_account_names=truststore-sa \
      bound_service_account_namespaces=keystore-site \
      policies=truststore-sa \
      ttl=24h

# Create patch to inject secrets
oc patch deployment truststore-updater --patch "$(cat <<EOF
spec:
   template:
      metadata:
        annotations:
           vault.hashicorp.com/agent-image: docker.io/hashicorp/vault
           vault.hashicorp.com/agent-inject: 'true'
           vault.hashicorp.com/role: 'truststore-sa'
           vault.hashicorp.com/agent-inject-template-vault-login.sh: |
             {{- with secret "secrets/data/truststore-updater/config" -}}
             vault login -method=userpass -token-only username={{ .Data.data.username }} password="{{ .Data.data.password }}"
             {{- end -}}
EOF
)"


### SETUP TRUSTSTORE-UPDATER SERVICE ###

# Download oc client and extract tar package.
https://mirror.openshift.com/pub/openshift-v4/clients/ocp/4.21.3/openshift-client-linux-arm64.tar.gz

# Build image
docker build -t ovekli/vault-kubernetes-client:1.0.0 -f oc-vault-client.dockerfile .

# Create deployment config
oc create namespace keystore-site
oc create sa truststore-sa
oc adm policy add-role-to-user edit -z truststore-sa
oc create -f truststore-updater-deployment.yml
