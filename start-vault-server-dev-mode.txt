# Start server
vault server -dev -dev-root-token-id root -dev-tls

# Export envs
export VAULT_ADDR='https://127.0.0.1:8200'
export VAULT_CACERT='/tmp/vault-tls617766039/vault-ca.pem'
Alternative:
export VAULT_TOKEN='<TOKEN>'

# Check status
vault status

# Login
vault login 
Token (will be hidden):

# List secrets
vault secrets list

# Create a token
vault token create --policy="default" --period=1h

# Lookup token
vault token lookup -accessor <accessor-id>

# vault auth userpass create one
vault auth enable userpass

# List all auth types (token, userpass...)
vault auth list

# Create policy
vault policy write developer-vault-policy - << EOF
path "dev-secrets/+/creds*" {
   capabilities = ["create", "list", "read", "update"]
}
EOF

# Create user
vault write /auth/userpass/users/danielle-vault-user \
    password='Flyaway Cavalier Primary Depose' \
    policies=developer-vault-policy

# Skapa en egen key/value (kv) secret
vault secrets enable -path=dev-secrets -version=2 kv

# Try login example 1
vault login -method=userpass username=danielle-vault-user

# Try login example 2
vault login \
    -no-print \
    -method=userpass \
    username=danielle-vault-user \
    password='Flyaway Cavalier Primary Depose'

# Lägg till en kv secret
vault kv put /dev-secrets/creds api-key=E6BED968-0FE3-411E-9B9B-C45812E4737A

# Fler tester på att lägga till kv secret
vault kv put /dev-secrets/creds-test api-token=xxxx crypto-token=yyyy

# Adderar en ny kv secret utan att ta bort befintliga
vault kv patch /dev-secrets/creds-test key-token=BBBB
