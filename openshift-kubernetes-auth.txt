# Enable /internal secret KV type
vault secrets enable -path=internal kv-v2

# Add an entry kv secret internal
vault kv put internal/database/config username="db-readonly-username" password="db-secret-password"

# List all secrets
vault secrets list

# List kv entries in path internal
vault kv list internal

# Get a specific kv entry (metadata/data)
vault kv get internal/database/config

# Add kubernetes authentication method
vault auth enable kubernetes

# Set address to the vault pod to parameter kubernetes_host
vault write auth/kubernetes/config \
      kubernetes_host="https://$KUBERNETES_PORT_443_TCP_ADDR:443"

# List authentications
vault auth list

# Verify kubernetes auth method
vault read auth/kubernetes/config

# Add read policy to secret /internal/data/database/config
vault policy write internal-app - <<EOF
path "internal/data/database/config" {
   capabilities = ["read"]
}
EOF

# List policys
vault policy list

# Read policy internal-app
vault policy read internal-app

# Create kubernetes auth role named internal-app
vault write auth/kubernetes/role/internal-app \
      bound_service_account_names=internal-app \
      bound_service_account_namespaces=default \
      policies=internal-app \
      ttl=24h

# List auth/kubernetes/roles
vault list auth/kubernetes/role

# Get role internal-app
vault read auth/kubernetes/role/internal-app

# Create a service account in openshift project
oc create sa internal-app

oc apply -f deployment-orgchart.yaml
